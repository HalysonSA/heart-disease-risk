# Base de regras SE-ENTÃO — CLASSIFICAÇÃO (has_heart_disease: 1 = com doença)
# Sintaxe completa em rules/_sintaxe.txt. Elaborada só com o conjunto de TREINO.
#
# Cada regra soma um peso ao fato "score" (pontuação de risco). A regra de decisão
# (R99, sempre a última) classifica como doente quem atinge score >= 0,5. O notebook 06
# usa "score" como pontuação contínua para as curvas ROC/PR.
# Percentuais entre parênteses = taxa de doença no treino (taxa geral = 30%).

# Padrão = classe majoritária no treino (69,7% sem doença).
PADRAO has_heart_disease = 0

# --- Sessão 1: sinais do teste de esforço (isquemia) ---
# angina no esforço (68% contra 19%); ST >= 1,5 (58%); FC máx <= 147 (76%); FC máx >= 184 (2%)
R01 | relacao    | SE exercise_induced_angina == 1 ENTAO score += 0.35
R02 | relacao    | SE st_depression >= 1.5 ENTAO score += 0.25
R03 | relacao    | SE max_heart_rate_achieved <= 147 ENTAO score += 0.3
R04 | relacao    | SE max_heart_rate_achieved >= 184 ENTAO score += -0.3

# --- Sessão 2: fatores de risco clássicos ---
# idade >= 65 (51%); HDL <= 45 (48%); LDL >= 126 (48%); PA sistólica >= 140 (48%); fumante atual (46%)
R05 | relacao    | SE age >= 65 ENTAO score += 0.15
R06 | relacao    | SE hdl <= 45 ENTAO score += 0.15
R07 | relacao    | SE ldl >= 126 ENTAO score += 0.15
R08 | relacao    | SE resting_bp_systolic >= 140 ENTAO score += 0.15
R09 | relacao    | SE smoker_Current == 1 ENTAO score += 0.15

# --- Sessão 3: ajustes finos ---
# diabetes: HbA1c >= 6,5 ou glicemia >= 140 (44-46%); IMC >= 29 (45%); histórico familiar (36%);
# jovem com HDL alto tem risco baixo (13% e 16%)
R10 | heuristica | SE hba1c >= 6.5 OU fasting_blood_sugar >= 140 ENTAO score += 0.1
R11 | heuristica | SE bmi >= 29 ENTAO score += 0.1
R12 | heuristica | SE family_history == 1 ENTAO score += 0.05
R13 | heuristica | SE age <= 43 E hdl >= 58 ENTAO score += -0.15

# --- Decisão ---
R99 | heuristica | SE score >= 0.5 ENTAO has_heart_disease = 1
